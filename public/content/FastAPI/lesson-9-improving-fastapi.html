Error Handling

from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse

app = FastAPI()

# Standard HTTPException
@app.get("/items/{item_id}")
def get_item(item_id: int):
    if item_id not in db:
        raise HTTPException(
            status_code=404,
            detail={"message": "Item not found", "item_id": item_id}
        )
    return db[item_id]

# Custom exception class
class AppError(Exception):
    def __init__(self, code: str, message: str):
        self.code = code
        self.message = message

# Custom exception handler
@app.exception_handler(AppError)
async def app_error_handler(request: Request, exc: AppError):
    return JSONResponse(
        status_code=400,
        content={"error_code": exc.code, "message": exc.message}
    )

# Override 404 globally
@app.exception_handler(404)
async def not_found_handler(request: Request, exc):
    return JSONResponse(status_code=404, content={"error": "Route not found"})

CORS Middleware

from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000", "https://myapp.com"],
    allow_credentials=True,
    allow_methods=["GET", "POST", "PUT", "PATCH", "DELETE"],
    allow_headers=["*"],
)

# Development only — allow all origins:
# allow_origins=["*"]
Never use allow_origins=["*"] in production

Wildcard origins disable CORS protection. In production, list specific origins. If you need credentials (cookies/auth), wildcard isn't even allowed by browsers — you must list origins explicitly.

Custom Middleware

import time
from fastapi import Request

@app.middleware("http")
async def add_timing_header(request: Request, call_next):
    start = time.time()
    response = await call_next(request)        # process request
    duration = time.time() - start
    response.headers["X-Process-Time"] = str(duration)
    return response

@app.middleware("http")
async def log_requests(request: Request, call_next):
    print(f"➡ {request.method} {request.url.path}")
    response = await call_next(request)
    print(f"⬅ {response.status_code}")
    return response

Background Tasks

from fastapi import BackgroundTasks
import smtplib

def send_welcome_email(email: str, name: str):
    # This runs after the response is sent — doesn't block the client
    print(f"Sending welcome email to {email}...")
    time.sleep(2)  # simulate slow email send
    print(f"Email sent to {name}!")

@app.post("/users", status_code=201)
def create_user(user: UserCreate, background_tasks: BackgroundTasks):
    # Create user in DB
    new_user = save_user(user)
    # Schedule background work
    background_tasks.add_task(send_welcome_email, user.email, user.name)
    # Response sent immediately, email sends asynchronously
    return new_user
Background tasks vs task queues

Background tasks are simple and great for fire-and-forget work (emails, logging). For reliable distributed task processing with retries and persistence, use a proper queue like Celery or BullMQ instead.

Dependency Injection

from fastapi import Depends

# A dependency function
def get_db():
    db = SessionLocal()   # create DB session
    try:
        yield db
    finally:
        db.close()         # always clean up

@app.get("/users/{user_id}")
def get_user(user_id: int, db = Depends(get_db)):
    return db.query(User).filter(User.id == user_id).first()

# Auth dependency
from fastapi import Header

def verify_api_key(x_api_key: str = Header(...)):
    if x_api_key != "secret123":
        raise HTTPException(status_code=403, detail="Invalid API Key")
    return x_api_key

@app.get("/protected", dependencies=[Depends(verify_api_key)])
def protected_route():
    return {"message": "You are authorized!"}
Dependencies are composable

Dependencies can depend on other dependencies, enabling a tree of reusable logic. Share DB sessions, auth checks, query pagination, and more — without duplicating code across routes.
