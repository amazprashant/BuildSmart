Load Model at Startup — Lifespan

from contextlib import asynccontextmanager
from fastapi import FastAPI
import joblib

# Global model variable (loaded once at startup)
ml_model = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Startup: load model into memory once
    global ml_model
    ml_model = joblib.load("model/iris_classifier.pkl")
    print("Model loaded ✅")
    yield
    # Shutdown: cleanup (optional)
    ml_model = None
    print("Model unloaded")

app = FastAPI(lifespan=lifespan)
Always load models in lifespan, not at module level

Loading models in the lifespan context manager ensures they are loaded once when the server starts — not once per request. Loading heavy models per-request would be catastrophically slow (100ms vs 10s+).

Prediction Endpoint (Tabular Data)

from pydantic import BaseModel
import numpy as np

class IrisInput(BaseModel):
    sepal_length: float
    sepal_width: float
    petal_length: float
    petal_width: float

class IrisPrediction(BaseModel):
    predicted_class: int
    class_name: str
    confidence: float

CLASS_NAMES = ["setosa", "versicolor", "virginica"]

@app.post("/predict", response_model=IrisPrediction)
def predict_iris(data: IrisInput):
    features = np.array([[
        data.sepal_length, data.sepal_width,
        data.petal_length, data.petal_width
    ]])

    prediction = ml_model.predict(features)[0]
    proba = ml_model.predict_proba(features)[0]
    confidence = float(proba[prediction])

    return IrisPrediction(
        predicted_class=int(prediction),
        class_name=CLASS_NAMES[prediction],
        confidence=round(confidence, 4)
    )

Image Prediction with UploadFile

from fastapi import UploadFile, File
from PIL import Image
import io

@app.post("/predict/image")
async def predict_image(file: UploadFile = File(...)):
    # Read image bytes
    contents = await file.read()
    image = Image.open(io.BytesIO(contents)).convert("RGB")

    # Preprocess for your model
    image = image.resize((224, 224))
    img_array = np.array(image) / 255.0          # normalize
    img_array = np.expand_dims(img_array, axis=0) # add batch dim

    predictions = ml_model.predict(img_array)
    class_idx = np.argmax(predictions[0])

    return {
        "class_index": int(class_idx),
        "confidence": float(predictions[0][class_idx]),
        "filename": file.filename
    }

Batch Predictions

from typing import List

class BatchInput(BaseModel):
    instances: List[IrisInput]

class BatchPrediction(BaseModel):
    predictions: List[IrisPrediction]
    count: int

@app.post("/predict/batch", response_model=BatchPrediction)
def batch_predict(batch: BatchInput):
    features = np.array([[
        inst.sepal_length, inst.sepal_width,
        inst.petal_length, inst.petal_width
    ] for inst in batch.instances])

    preds = ml_model.predict(features)
    probas = ml_model.predict_proba(features)

    results = [
        IrisPrediction(
            predicted_class=int(p),
            class_name=CLASS_NAMES[p],
            confidence=round(float(probas[i][p]), 4)
        )
        for i, p in enumerate(preds)
    ]

    return BatchPrediction(predictions=results, count=len(results))
Async vs sync prediction routes

Use async def for I/O-heavy operations (image reading, HTTP calls). For CPU-heavy model inference (sklearn, PyTorch), use regular def — FastAPI runs sync route handlers in a thread pool, keeping the event loop free.

Best Practices

🚀 Load Once

Load models in lifespan, keep in memory as globals or app state.

📝 Typed I/O

Use Pydantic for input validation and response_model for consistent output.

⚡ Batch

Support batch prediction endpoints — single model call is much faster than N calls.

🛡 Validate

Validate input ranges (no negative image sizes, reasonable feature values) at the Pydantic layer.
